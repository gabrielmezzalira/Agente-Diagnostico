      def _decrypt_pricing_key(db: Client, project_id: str) -> str:
          """Busca e descriptografa pricing_api_key do Vault para o projeto."""
          proj = db.table("projects").select("pricing_llm_provider, pricing_llm_model, pricing_api_key_secret_id").eq("id", project_id).execute()
          if not proj.data:
              raise HTTPException(status_code=404, detail="Project not found")
          row = proj.data[0]
          secret_id = row.get("pricing_api_key_secret_id")
          if not secret_id:
              raise HTTPException(status_code=422, detail="Chave de API do Precificador não configurada neste projeto. Configure em Configurações do Projeto.")
          decrypted = db.table("vault.decrypted_secrets").select("secret").eq("id", str(secret_id)).execute()
          if not decrypted.data:
              raise HTTPException(status_code=500, detail="Falha ao descriptografar chave de API do Precificador.")
          return row.get("pricing_llm_provider") or "", row.get("pricing_llm_model") or "", decrypted.data[0]["secret"]

    Endpoint 1 — import-from-diagnosis:
      @router.post("/pricings/{pricing_id}/import-from-diagnosis", response_model=list[PricingFeatureResponse], status_code=status.HTTP_201_CREATED)
      async def import_from_diagnosis(pricing_id: UUID, db: Client = Depends(get_supabase)) -> list[PricingFeatureResponse]:
          repo = PricingRepository(db)

          if not pricing:

          project_id = str(pricing["project_id"])

          svc = LLMPricingService(repo, llm)
          return svc.import_from_diagnosis(str(pricing_id), project_id)
