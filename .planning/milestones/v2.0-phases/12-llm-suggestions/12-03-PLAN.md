      }

    Export function useLLMSuggestions(
      pricingId: string | undefined,
      setFeatures: React.Dispatch<React.SetStateAction<PricingFeature[]>>
    ):
      Estado interno:
        - importing: boolean (false)
        - importError: string | null (null)
        - suggesting: boolean (false)
        - suggestError: string | null (null)
        - suggestions: SuggestionWithId[] ([])
        - showSuggestions: boolean (false)

      Método importFromDiagnosis: useCallback(async () => {
        1. if (!pricingId) return
        2. setImporting(true); setImportError(null)
        3. try { const newFeatures = await api.pricings.importFromDiagnosis(pricingId) }
        4. setFeatures(fs => [...fs, ...newFeatures])
        5. catch (e: unknown) { setImportError(e instanceof Error ? e.message : 'Erro ao importar do diagnóstico') }
        6. finally { setImporting(false) }
      }, [pricingId, setFeatures])

      Método suggestFeatures: useCallback(async () => {
        1. if (!pricingId) return
        2. setSuggesting(true); setSuggestError(null)
        3. try { const raw = await api.pricings.suggestFeatures(pricingId) }
        4. setSuggestions(raw.map(s => ({ ...s, _localId: Math.random().toString(36).slice(2) })))
        5. setShowSuggestions(true)
        6. catch (e: unknown) { setSuggestError(e instanceof Error ? e.message : 'Erro ao sugerir funcionalidades') }
        7. finally { setSuggesting(false) }
      }, [pricingId])

      Método acceptSuggestion: useCallback(async (localId: string, addFeature: (body: object) => Promise<void>) => {
        1. const s = suggestions.find(s => s._localId === localId); if (!s) return
        2. await addFeature({ bloco: s.bloco, funcionalidade: s.funcionalidade, horas: Number(s.horas), citi_responsible: true })
        3. setSuggestions(prev => prev.filter(s => s._localId !== localId))
      }, [suggestions])

      Método rejectSuggestion: useCallback((localId: string) => {
        setSuggestions(prev => prev.filter(s => s._localId !== localId))
      }, [])

      Método closeSuggestions: useCallback(() => {
        setShowSuggestions(false)
        setSuggestions([])
      }, [])
      — closeSuggestions DEVE ser definido com useCallback e dep array vazio []. Isso garante
        que a referência é estável entre re-renders, permitindo que o useEffect em
        PricingEditorPage inclua closeSuggestions no dep array sem causar loops infinitos.

      Retorno do hook:
        { importing, importError, suggesting, suggestError, suggestions, showSuggestions,
          importFromDiagnosis, suggestFeatures, acceptSuggestion, rejectSuggestion, closeSuggestions }
